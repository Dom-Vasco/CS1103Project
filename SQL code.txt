CREATE TABLE Passenger (
    Passenger_ID INT PRIMARY KEY,
    Name         VARCHAR(100),
    Passport_num VARCHAR(20),
    Email        VARCHAR(100)
);
CREATE TABLE Flight (
    Flight_ID      INT PRIMARY KEY,
    Airplane_ID    INT,
    Des_Airport    VARCHAR(100),
    Or_Airport     VARCHAR(100),
    Departure_date DATE,
    Arrival_date   DATE
);
CREATE TABLE Transact (
    Payment_ID     INT PRIMARY KEY,
    Date           DATE,
    Payment_method VARCHAR(50),
    Passenger_ID   INT,
    FOREIGN KEY (Passenger_ID) REFERENCES Passenger(Passenger_ID)
);

CREATE TABLE Reservation (
    Booking_ID     INT PRIMARY KEY,
    Passenger_name VARCHAR(100),
    Date           DATE,
    Destination    VARCHAR(100)
);
CREATE TABLE Ticket (
    Ticket_ID      INT PRIMARY KEY,
    Passenger_name VARCHAR(100),
    Seat_num       VARCHAR(10),
    Class          VARCHAR(20),
    Flight_ID      INT,
    FOREIGN KEY (Flight_ID) REFERENCES Flight(Flight_ID)
);

CREATE TABLE Passenger_Flight (
    Passenger_ID  INT,
    Flight_ID     INT,
    Booking_date  DATE,
    PRIMARY KEY (Passenger_ID, Flight_ID),
    FOREIGN KEY (Passenger_ID) REFERENCES Passenger(Passenger_ID),
    FOREIGN KEY (Flight_ID)    REFERENCES Flight(Flight_ID)
);

CREATE TABLE Passenger_Reservation (
    Passenger_ID  INT,
    Booking_ID    INT,
    PRIMARY KEY (Passenger_ID, Booking_ID),
    FOREIGN KEY (Passenger_ID) REFERENCES Passenger(Passenger_ID),
    FOREIGN KEY (Booking_ID)   REFERENCES Reservation(Booking_ID) );
    
CREATE TABLE Reservation_Ticket (
    Booking_ID  INT,
    Ticket_ID   INT,
    PRIMARY KEY (Booking_ID, Ticket_ID),
    FOREIGN KEY (Booking_ID) REFERENCES Reservation(Booking_ID),
    FOREIGN KEY (Ticket_ID)  REFERENCES Ticket(Ticket_ID));
